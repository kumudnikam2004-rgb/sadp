PizzaStore

abstract class Pizza {
    String name;
    void prepare() { System.out.println("Preparing " + name); }
    void bake() { System.out.println("Baking " + name); }
    void cut() { System.out.println("Cutting " + name); }
    void box() { System.out.println("Boxing " + name); }
}

class NYStyleCheesePizza extends Pizza {
    public NYStyleCheesePizza() { name = "NY Style Cheese Pizza"; }
}

class ChicagoStyleCheesePizza extends Pizza {
    public ChicagoStyleCheesePizza() { name = "Chicago Style Cheese Pizza"; }
}

abstract class PizzaStore {
    // The Factory Method
    protected abstract Pizza createPizza(String type);

    public Pizza orderPizza(String type) {
        Pizza pizza = createPizza(type); // Uses the factory method
        pizza.prepare();
        pizza.bake();
        pizza.cut();
        pizza.box();
        return pizza;
    }
}

class NYPizzaStore extends PizzaStore {
    protected Pizza createPizza(String type) {
        return new NYStyleCheesePizza(); 
    }
}

class ChicagoPizzaStore extends PizzaStore {
    protected Pizza createPizza(String type) {
        return new ChicagoStyleCheesePizza(); 
    }
}

public class PizzaMain {
    public static void main(String[] args) {
        PizzaStore nyStore = new NYPizzaStore();
        nyStore.orderPizza("cheese");
        
        System.out.println("---");
        
        PizzaStore chicagoStore = new ChicagoPizzaStore();
        chicagoStore.orderPizza("cheese");
    }
}



Singleton for Multithreading

class ThreadSafeSingleton {
    private static ThreadSafeSingleton instance;

    // 1. Private constructor
    private ThreadSafeSingleton() {
        System.out.println("Singleton Instance Created!");
    }

    // 2. Synchronized method for thread safety
    public static synchronized ThreadSafeSingleton getInstance() {
        if (instance == null) {
            instance = new ThreadSafeSingleton();
        }
        return instance;
    }
}

public class SingletonMain {
    public static void main(String[] args) {
        // Creating two threads to test thread safety
        Thread t1 = new Thread(() -> ThreadSafeSingleton.getInstance());
        Thread t2 = new Thread(() -> ThreadSafeSingleton.getInstance());

        t1.start();
        t2.start();
        // You will only see "Singleton Instance Created!" printed once.
    }
}

Lab 1: I/O Decorator (Uppercase to Lowercase)

import java.io.*;

class LowerCaseInputStream extends FilterInputStream {
    public LowerCaseInputStream(InputStream in) { super(in); }

    @Override
    public int read() throws IOException {
        int c = super.read();
        return (c == -1 ? c : Character.toLowerCase((char) c));
    }
}

public class DecoratorMain {
    public static void main(String[] args) throws IOException {
        String text = "HELLO WORLD!";
        InputStream in = new LowerCaseInputStream(new ByteArrayInputStream(text.getBytes()));
        
        int c;
        while ((c = in.read()) >= 0) {
            System.out.print((char) c);
        }
        // Output: hello world!
    }
}


Lab 2: Adapter Pattern (Enumeration to Iterator)

import java.util.*;

// The Adapter
class EnumerationIterator<T> implements Iterator<T> {
    private Enumeration<T> enumeration;

    public EnumerationIterator(Enumeration<T> enumeration) {
        this.enumeration = enumeration;
    }

    public boolean hasNext() { return enumeration.hasMoreElements(); }
    public T next() { return enumeration.nextElement(); }
}

public class AdapterMain {
    public static void main(String[] args) {
        Vector<String> days = new Vector<>(Arrays.asList("Mon", "Tue", "Wed"));
        
        // Wrap the Enumeration in our Adapter
        Iterator<String> iterator = new EnumerationIterator<>(days.elements());
        
        while (iterator.hasNext()) {
            System.out.println(iterator.next());
        }
    }
}


Lab 3: Bridge Pattern (Produce and Assemble Vehicles)

// Implementor
interface Workshop { void work(); }

class Produce implements Workshop {
    public void work() { System.out.print("Produced "); }
}
class Assemble implements Workshop {
    public void work() { System.out.println("and Assembled."); }
}

// Abstraction
abstract class Vehicle {
    protected Workshop work1, work2;
    protected Vehicle(Workshop w1, Workshop w2) { work1 = w1; work2 = w2; }
    abstract void manufacture();
}

class Car extends Vehicle {
    public Car(Workshop w1, Workshop w2) { super(w1, w2); }
    void manufacture() {
        System.out.print("Car: ");
        work1.work(); work2.work();
    }
}

public class BridgeMain {
    public static void main(String[] args) {
        Vehicle myCar = new Car(new Produce(), new Assemble());
        myCar.manufacture(); 
        // Output: Car: Produced and Assembled.
    }
}

Extra Practice 2: Proxy Pattern (Image Loading)

// Subject
interface Image { void display(); }

// Real Subject (Heavy to load)
class RealImage implements Image {
    private String filename;
    public RealImage(String filename) {
        this.filename = filename;
        System.out.println("Loading from disk: " + filename);
    }
    public void display() { System.out.println("Displaying: " + filename); }
}

// Proxy (Lightweight)
class ProxyImage implements Image {
    private String filename;
    private RealImage realImage;

    public ProxyImage(String filename) { this.filename = filename; }

    public void display() {
        if (realImage == null) {
            realImage = new RealImage(filename); // Lazy initialization
        }
        realImage.display();
    }
}

public class ProxyMain {
    public static void main(String[] args) {
        Image img = new ProxyImage("photo.jpg");
        System.out.println("Image object created, but not loaded yet.");
        
        img.display(); // Loads and displays
        img.display(); // Only displays (already loaded)
    }
}

Lab 1: Observer Pattern (Weather Station)

import java.util.Observable;
import java.util.Observer;

// 1. The Subject (Observable)
class WeatherData extends Observable {
    private float temperature, humidity, pressure;

    public void measurementsChanged() {
        setChanged();
        notifyObservers();
    }

    public void setMeasurements(float temp, float humidity, float pressure) {
        this.temperature = temp;
        this.humidity = humidity;
        this.pressure = pressure;
        measurementsChanged();
    }

    public float getTemperature() { return temperature; }
    public float getHumidity() { return humidity; }
    public float getPressure() { return pressure; }
}

// 2. The Observer
class CurrentConditionsDisplay implements Observer {
    public void update(Observable obs, Object arg) {
        if (obs instanceof WeatherData) {
            WeatherData w = (WeatherData) obs;
            System.out.println("Current conditions: " + w.getTemperature() + 
                               "F degrees and " + w.getHumidity() + "% humidity");
        }
    }
}

public class ObserverMain {
    public static void main(String[] args) {
        WeatherData weatherData = new WeatherData();
        CurrentConditionsDisplay display = new CurrentConditionsDisplay();
        
        weatherData.addObserver(display);
        weatherData.setMeasurements(80, 65, 30.4f);
    }
}


Lab 2: Command Pattern with Undo (Ceiling Fan)

// 1. Command Interface
interface Command {
    void execute();
    void undo();
}

// 2. Receiver
class CeilingFan {
    int speed = 0;
    void high() { speed = 3; System.out.println("Fan is HIGH"); }
    void off() { speed = 0; System.out.println("Fan is OFF"); }
}

// 3. Concrete Command
class FanHighCommand implements Command {
    CeilingFan fan;
    int prevSpeed;

    public FanHighCommand(CeilingFan fan) { this.fan = fan; }

    public void execute() {
        prevSpeed = fan.speed;
        fan.high();
    }

    public void undo() {
        if (prevSpeed == 0) fan.off();
        else fan.high();
    }
}

// 4. Invoker
public class FanCommandMain {
    public static void main(String[] args) {
        CeilingFan fan = new CeilingFan();
        Command fanHigh = new FanHighCommand(fan);

        fanHigh.execute(); // Turns fan High
        fanHigh.undo();    // Reverts to Off
    }
}

Extra Practice 1: State Pattern (Gumball Machine)

// 1. State Interface
interface State {
    void insertQuarter();
    void ejectQuarter();
    void turnCrank();
    void dispense();
}

// 2. Context Object
class GumballMachine {
    State noQuarter, hasQuarter, currentState;

    public GumballMachine() {
        noQuarter = new NoQuarterState(this);
        hasQuarter = new HasQuarterState(this);
        currentState = noQuarter;
    }

    void setState(State state) { this.currentState = state; }
    
    public void insertQuarter() { currentState.insertQuarter(); }
    public void ejectQuarter() { currentState.ejectQuarter(); }
    public void turnCrank() { 
        currentState.turnCrank(); 
        currentState.dispense(); 
    }
}

// 3. Concrete States
class NoQuarterState implements State {
    GumballMachine gm;
    public NoQuarterState(GumballMachine gm) { this.gm = gm; }
    public void insertQuarter() { System.out.println("Quarter inserted."); gm.setState(gm.hasQuarter); }
    public void ejectQuarter() { System.out.println("You haven't inserted a quarter."); }
    public void turnCrank() { System.out.println("You turned, but there's no quarter."); }
    public void dispense() { System.out.println("You need to pay first."); }
}

class HasQuarterState implements State {
    GumballMachine gm;
    public HasQuarterState(GumballMachine gm) { this.gm = gm; }
    public void insertQuarter() { System.out.println("You can't insert another quarter."); }
    public void ejectQuarter() { System.out.println("Quarter returned."); gm.setState(gm.noQuarter); }
    public void turnCrank() { System.out.println("You turned..."); }
    public void dispense() { System.out.println("A gumball comes rolling out!"); gm.setState(gm.noQuarter); }
}

public class StateMain {
    public static void main(String[] args) {
        GumballMachine machine = new GumballMachine();
        machine.insertQuarter();
        machine.turnCrank();
    }
}

package com.example.demo;

import java.util.ArrayList;
import java.util.List;

import org.springframework.web.bind.annotation.*;

@RestController
@RequestMapping("/employees")
public class HRController {

    List<Employee> employees = new ArrayList<>();

    @PostMapping
    public String addEmployee(@RequestBody Employee e) {
        employees.add(e);
        return "Employee added successfully";
    }

    @GetMapping
    public List<Employee> getEmployees() {
        return employees;
    }

    @GetMapping("/{id}")
    public Employee getEmployee(@PathVariable int id) {

        for (Employee e : employees) {
            if (e.getId() == id) {
                return e;
            }
        }

        return null;
    }

    @DeleteMapping("/{id}")
    public String deleteEmployee(@PathVariable int id) {

        for (Employee e : employees) {
            if (e.getId() == id) {
                employees.remove(e);
                return "Employee deleted successfully";
            }
        }

        return "Employee not found";
    }
}

#UserController.java
package com.example.demo;

import java.util.ArrayList;
import java.util.List;

import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.bind.annotation.RestController;

@RestController
@RequestMapping("/users")
public class UserController {

    List<User> users = new ArrayList<>();

    @PostMapping("/register")
    public String register(
            @RequestParam String name,
            @RequestParam String email,
            @RequestParam String username) {

        // Validation
        if (name.isEmpty()) {
            return "Name cannot be empty";
        }

        if (email.isEmpty() || !email.contains("@")) {
            return "Invalid email";
        }

        if (username.isEmpty()) {
            return "Username cannot be empty";
        }

        // Check username already exists
        for (User u : users) {
            if (u.getUsername().equals(username)) {
                return "Username already exists";
            }
        }

        User user = new User(name, email, username);
        users.add(user);

        return "Registration successful";
    }

    @PostMapping("/login")
    public String login(@RequestParam String username) {

        for (User u : users) {
            if (u.getUsername().equals(username)) {
                return "Login successful";
            }
        }

        return "User not found";
    }

    @GetMapping
    public List<User> getUsers() {
        return users;
    }
}

# EmployeeController.java

package com.example.demo;

import java.io.FileWriter;
import java.io.IOException;

import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.bind.annotation.RestController;

@RestController
@RequestMapping("/employees")
public class EmployeeController {

    @PostMapping("/save")
    public String saveEmployee(
            @RequestParam int eno,
            @RequestParam String ename,
            @RequestParam String designation,
            @RequestParam String deptName,
            @RequestParam double salary) {

        try {

            FileWriter file = new FileWriter("employees.txt", true);

            file.write(
                "Employee No: " + eno +
                ", Name: " + ename +
                ", Designation: " + designation +
                ", Department: " + deptName +
                ", Salary: " + salary +
                "\n"
            );

            file.close();

            return "Employee details saved successfully";

        } catch (IOException e) {

            return "Error while saving employee";
        }
    }
}

package com.example.demo;

import java.util.ArrayList;
import java.util.List;

import org.springframework.web.bind.annotation.*;

@RestController
@RequestMapping("/erp")
public class ERPController {

    List<Inventory> inventory = new ArrayList<>();
    List<Sales> sales = new ArrayList<>();

    // Add inventory
    @PostMapping("/inventory")
    public String addInventory(
            @RequestParam int id,
            @RequestParam String product,
            @RequestParam int quantity) {

        Inventory item = new Inventory(id, product, quantity);
        inventory.add(item);

        return "Inventory added successfully";
    }

    // View inventory
    @GetMapping("/inventory")
    public List<Inventory> getInventory() {
        return inventory;
    }

    // Add sale
    @PostMapping("/sales")
    public String addSale(
            @RequestParam int id,
            @RequestParam String product,
            @RequestParam int quantity,
            @RequestParam double amount) {

        Sales sale = new Sales(id, product, quantity, amount);
        sales.add(sale);

        return "Sale added successfully";
    }

    // View sales
    @GetMapping("/sales")
    public List<Sales> getSales() {
        return sales;
    }

    // Calculate total sales
    @GetMapping("/finance")
    public String getFinance() {

        double total = 0;

        for (Sales sale : sales) {
            total = total + sale.getAmount();
        }

        return "Total Sales Amount = " + total;
    }
}

package com.example.demo;

import java.util.ArrayList;
import java.util.List;

import org.springframework.web.bind.annotation.*;

@RestController
@RequestMapping("/inventory")
public class InventoryController {

    List<Product> products = new ArrayList<>();

    // Add product
    @PostMapping
    public String addProduct(
            @RequestParam int id,
            @RequestParam String name,
            @RequestParam int quantity,
            @RequestParam double price) {

        Product product = new Product(id, name, quantity, price);
        products.add(product);

        return "Product added successfully";
    }

    // View all products
    @GetMapping
    public List<Product> getProducts() {
        return products;
    }

    // Find product by ID
    @GetMapping("/{id}")
    public Product getProduct(@PathVariable int id) {

        for (Product product : products) {
            if (product.getId() == id) {
                return product;
            }
        }

        return null;
    }

    // Update quantity
    @PutMapping("/{id}")
    public String updateQuantity(
            @PathVariable int id,
            @RequestParam int quantity) {

        for (Product product : products) {

            if (product.getId() == id) {
                product.setQuantity(quantity);
                return "Quantity updated successfully";
            }
        }

        return "Product not found";
    }

    // Delete product
    @DeleteMapping("/{id}")
    public String deleteProduct(@PathVariable int id) {

        for (Product product : products) {

            if (product.getId() == id) {
                products.remove(product);
                return "Product deleted successfully";
            }
        }

        return "Product not found";
    }
}
